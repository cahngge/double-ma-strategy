初始资金 = 100000
资金 = 初始资金
持股 = 0
资产记录 = []

for i in range(len(df)):
    日期 = df.iloc[i]["日期"]
    收盘价 = df.iloc[i]["收盘"]
    信号 = df.iloc[i]["信号"]
    
    if 信号 == "持仓" and 持股 == 0:
        持股 = 资金 / 收盘价
        资金 = 0
    elif 信号 == "空仓" and 持股 > 0:
        资金 = 持股 * 收盘价
        持股 = 0
    
    总资产 = 资金 + 持股 * 收盘价
    资产记录.append(总资产)

df["总资产"] = 资产记录
最终资产 = 资产记录[-1]
收益率 = (最终资产 - 初始资金) / 初始资金 * 100

print(f"初始资金：{初始资金}元")
print(f"最终资产：{最终资产:.2f}元")
print(f"收益率：{收益率:.2f}%")
